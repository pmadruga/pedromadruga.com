{{- /* Image render hook with modern format support and responsive images */ -}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- if $img -}} {{- /* Define responsive widths */ -}} {{- $widths := slice 400 800 1200 1600 -}} {{- /* Get image dimensions */ -}} {{- $width := $img.Width -}} {{- $height := $img.Height -}} {{- /* Collect candidates first so the srcset has no trailing comma and is omitted entirely for images narrower than the smallest width */ -}} {{- $candidates := slice -}} {{- range $widths -}} {{- if le . $width -}} {{- $resized := $img.Resize (printf "%dx webp q85" .) -}} {{- $candidates = $candidates | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- /* Images between ladder steps (e.g. 560px or 1300px wide) would only get the smaller step and render upscaled on retina — add the native width as the top candidate. Wider than the ladder, 1600w is plenty. */ -}} {{- if and (lt $width 1600) (not (in $widths $width)) -}} {{- $native := $img.Resize (printf "%dx webp q85" $width) -}} {{- $candidates = $candidates | append (printf "%s %dw" $native.RelPermalink $width) -}} {{- end -}} {{- with $candidates }} {{- end }} {{- /* Fallback to original format. The first in-content image is only an LCP candidate on posts without a cover (with one, the cover is the LCP element and gets fetchpriority=high in cover.html); even then it is usually below the fold, so it gets eager loading but no priority hint, which would make it compete with the fonts for the title paint. */ -}} {{- else -}} {{- /* External image or resource not found - use simple img tag */ -}} {{- end -}}
