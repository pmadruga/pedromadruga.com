{{- /* Year-grouped list of blog posts — the shared body of both the /blog/ archive (layouts/archives.html) and the "Archive" section on /newsletter/ (layouts/newsletter.html), so the two stay in sync while they run side by side. Params: "yearLevel"/"titleLevel" set the heading tags, since the two pages sit at different depths — /blog/ puts the years directly under its h1, while the newsletter page nests them under an h2. Both default to the /blog/ shape. No month grouping: each entry is one row — full date | title | reading time. Row layout lives in assets/css/extended/custom_css1.css (all class-based, so the heading tag makes no visual difference). The heading tags go through printf/safeHTML because Go's html/template escapes a tag name built from a variable (`<{{ $x }}>` renders as literal <h2>). The values are template-local literals, never user input. */ -}} {{- $yearLevel := .yearLevel | default "h2" }} {{- $titleLevel := .titleLevel | default "h3" }} {{- $pages := where site.RegularPages "Type" "in" site.Params.mainSections }} {{- range $pages.GroupByPublishDate "2006" }} {{- if ne .Key "0001" }}
{{- $year := replace .Key "0001" "" }} {{- printf `<%s class="archive-year-header" id="%s">` $yearLevel $year | safeHTML }} {{- $year -}} {{- printf `` $yearLevel | safeHTML }}
{{- range .Pages }} {{- if eq .Kind "page" }}
{{- .Date | time.Format (default ":date_long" site.Params.DateFormat) -}} {{- printf `<%s class="archive-entry-title">` $titleLevel | safeHTML }} {{- .Title | markdownify }} {{- if .Draft }} {{- end }} {{- printf `` $titleLevel | safeHTML }} {{- if (.Param "ShowReadingTime") }} {{ i18n "read_time" .ReadingTime | default (printf "%d min" .ReadingTime) }} {{- end }}
{{- end }} {{- end }}
{{- end }} {{- end }}
