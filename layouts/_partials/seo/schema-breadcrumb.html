{{- /* JSON-LD breadcrumb structured data */ -}} {{- if not .IsHome -}} {{- $scratch := newScratch -}} {{- $scratch.Set "position" 1 -}} {{- $scratch.Set "breadcrumbs" slice -}} {{- /* Add home as first item */ -}} {{- $scratch.Add "breadcrumbs" (dict "position" 1 "name" "Home" "url" .Site.BaseURL) -}} {{- $scratch.Set "position" 2 -}} {{- /* Add section if exists */ -}} {{- with .Section -}} {{- $sectionURL := printf "%s/%s/" $.Site.BaseURL . -}} {{- $scratch.Add "breadcrumbs" (dict "position" ($scratch.Get "position") "name" (. | title) "url" $sectionURL) -}} {{- $scratch.Set "position" (add ($scratch.Get "position") 1) -}} {{- end -}} {{- /* Add current page */ -}} {{- if not .IsSection -}} {{- $scratch.Add "breadcrumbs" (dict "position" ($scratch.Get "position") "name" .Title "url" .Permalink) -}} {{- end -}} {{- end -}}
